// TRH FASE 4 FIX - FUTURO BUENO: tinte VERDE fuerte (R*1/2, G, B*1/4).
// El tinte antiguo (R*3/4, B*3/4) dejaba el cielo AZUL OSCURO porque el
// azul seguia dominando. Con B*1/4 el verde pasa a dominar en TODOS los
// colores (cielo -> verde, hierba -> sigue verde). Se aplica a los 8 banks.
function TRH_GreenTint
	// TRH FIX (QUITAR VERDE RADIOACTIVO): tinte SUAVE de atardecer -
	//   R*3/4, G, B*3/4 (pedido explicito: en vez de R*1/2, G, B*1/4
	//   que dejaba el cielo verde tipo Chernobil).
	temp0 = 0
	while temp0 < 8
		temp1 = 0
		while temp1 < 256
			GetPaletteEntry(temp0, temp1, temp2)
			// R' = R*3/4 (bits 16-23)
			temp3 = temp2
			temp3 &= 0xFF0000
			temp3 >>= 16
			temp3 *= 3
			temp3 /= 4
			temp3 <<= 16
			// B' = B*3/4 (bits 0-7)
			temp4 = temp2
			temp4 &= 0x0000FF
			temp4 *= 3
			temp4 /= 4
			// G intacto (bits 8-15)
			temp5 = temp2
			temp5 &= 0x00FF00
			temp3 += temp5
			temp3 += temp4
			SetPaletteEntry(temp0, temp1, temp3)
			temp1++
		loop
		temp0++
	loop
end function
