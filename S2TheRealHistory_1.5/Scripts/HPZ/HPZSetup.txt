function HPZSetup_Function97

	object.value9 >>= 16
	object.value10 = temp1
	object.value10 >>= 8
	object.value10 &= 255
	object.value11 = temp1
	object.value11 &= 255
	object.value12 = temp2
	object.value12 >>= 16
	object.value13 = temp2
	object.value13 >>= 8
	object.value13 &= 255
	object.value14 = temp2
	object.value14 &= 255
	if object.value12 == 0
		object.value12 = 1
	end if
	if object.value13 == 0
		object.value13 = 1
	end if
	if object.value14 == 0
		object.value14 = 1
	end if
	object.value9 <<= 8
	object.value10 <<= 8
	object.value11 <<= 8
	object.value9 /= object.value12
	object.value10 /= object.value13
	object.value11 /= object.value14
	if object.value9 == 0
		object.value9 = 256
	end if
	if object.value10 == 0
		object.value10 = 256
	end if

function HPZSetup_Function103
	switch stage.actNum
		case 1 //Taxman
			// TRH: Zone28 (Warp Point) y Olympus (Zone08, listPos 20) NO deben
			// rebotar a titulo. Nota: saveRAM[481] nunca se escribe en todo el
			// mod ni en la base => "if saveRAM[481] == 0" es SIEMPRE TRUE, asi
			// que sin estas guardas Olympus rebotaba al titulo en el arranque
			// (sintoma: "Olympus no carga"). PARTE 11: guarda listPos 20 anadida.
			if stage.listPos != 46
				if stage.listPos != 35
					if saveRAM[481] == 0
						CallFunction(StageSetup_LoadStageState)
						stage.activeList = PRESENTATION_STAGE
						stage.listPos = 0
						// TRH FIX4: starPostID = 0 ANTES de cargar.
						starPostID = 0
						LoadStage()
					end if
				end if
			end if
			break
		case 2 //Proto
			stage.listPos = specialStage.nextZone
			stage.activeList = REGULAR_STAGE
			if recGoggles == 2		
				recGoggles = 1
				starPostID = 428				
			else
				recGoggles = 0
				CallFunction(StageSetup_LoadStageState)
				stage.activeList = PRESENTATION_STAGE
				stage.listPos = 0
				// TRH FIX4: starPostID = 0 ANTES de cargar.
				starPostID = 0
				LoadStage()
			end if
			break
	end switch
end function
