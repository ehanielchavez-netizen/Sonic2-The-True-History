function CCZSetup_Function101
	CheckEqual(object[25].type, TypeName[Music Event])
	temp0 = checkResult
	CheckEqual(object[25].propertyValue, 2)
	temp0 &= checkResult
	CheckEqual(stage.musicFlag, 0)
	temp0 &= checkResult
	if temp0 == 0
		switch music.currentTrack
		case 0
			SetMusicTrack("Invincibility_F.ogg", 2, 0x76E4)
			SwapMusicTrack("GenocideDeathEgg.ogg", 0, 0x7AA16, 0x1EF0)
			break
		case 2
			SetMusicTrack("GenocideDeathEgg.ogg", 0, 0x7AA16)
			SwapMusicTrack("Invincibility_F.ogg", 2, 0x76E4, 0x1F40)
			break
		default
			SetMusicTrack("GenocideDeathEgg.ogg", 0, 0x7AA16)
			SetMusicTrack("Invincibility_F.ogg", 2, 0x76E4)
			break
		end switch
	else
		stage.musicFlag = 1
	end if
end function

event ObjectMain
	if stage.actNum == 2
		if object[0].xpos > 0xE800000
			stage.newWaterLevel = 0x6000000
		else
			stage.newWaterLevel = 0x6000000
		end if
	end if
	object.value1++
	if object.value1 == 5
		object.value1 = 0
		RotatePalette(0, 167, 169, 2)
	end if
	object.value1++
	if object.value1 == 5
		object.value1 = 0
		RotatePalette(0, 211, 214, 1)
	end if
	object.value1++
	if object.value1 == 5
		object.value1 = 0
		RotatePalette(0, 180, 182, 1)
	end if
	if CCZSetup_value33 > 0
		CCZSetup_value33--
	end if
	temp0 = object.value8
	temp0 /= 6
	GetTableValue(CCZSetup_value35, temp0, CCZSetup_table42)
	object.value8++
	object.value8 %= 84
	// TRH: waterslide automatico desactivado en Genocide City
	// Evita el bucle de muerte al reaparecer en Star Posts sobre tiles de angulo 1
	if object[0].state == PlayerObject_Waterslide
		if object.value16 == 0
			if object.value17 == 0
				PlaySfx(SfxName[Waterfall], 0)
				StopSfx(SfxName[Waterfall Loop])
				object.value17 = 1
			else
				StopSfx(SfxName[Waterfall])
				PlaySfx(SfxName[Waterfall Loop], 0)
			end if
		end if
		object.value16++
		object.value16 &= 63
	else
		if object.value16 != 0
			object.value16++
			object.value16 &= 63
		else
			object.value16 = 0
			object.value17 = 0
		end if
	end if
	if CCZSetup_value36 == 0
		if stage.debugMode == 0
			if options.stageSelectFlag == 0
				if object[30].type == TypeName[Act Finish]
					CCZSetup_value36 = 1
					//CallNativeFunction2(SetAchievement, 7, 100)
					//achievement.trigger = 8 //A Secret Revealed
				end if
			end if
		end if
	end if
