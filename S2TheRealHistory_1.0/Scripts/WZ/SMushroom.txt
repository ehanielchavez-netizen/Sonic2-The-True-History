private alias 46 : TYPE_SmallMushroom

// Function declarations
reserve function SmallMushroom_DebugDraw
reserve function SmallMushroom_DebugSpawn

// Static Values

// Tables

function SmallMushroom_DebugDraw
	temp0 = object.direction
	temp0 &= 1
	DrawSprite(temp0)
end function


function SmallMushroom_DebugSpawn
	CreateTempObject(TypeName[SmallMushroom], 0, object.xpos, object.ypos)
	object[tempObjectPos].drawOrder = 4
	object[tempObjectPos].propertyValue = object.direction
	object[tempObjectPos].propertyValue &= 1
end function


event ObjectDraw
	DrawSprite(0)
end event

	CheckCurrentStageFolder("Zone13")
	if checkResult == 1
		LoadSpriteSheet("WZ/Objects.gif")

		//   hongos invisibles. Ese frame vive en Objects2.gif (512x512),
		//   asi que aqui se carga esa hoja.
		LoadSpriteSheet("WZ/Objects2.gif")
		SpriteFrame(-8, -8, 16, 16, 111, 333)
