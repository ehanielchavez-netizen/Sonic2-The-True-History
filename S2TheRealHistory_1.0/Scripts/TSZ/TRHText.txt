reserve function TRHText_DrawLine


event ObjectMain
	if object.value0 < 360
		object.value0++
	else
		// se autodestruye tras mostrarse unos segundos
		ResetObjectEntity(object.entityPos, TypeName[Blank Object], 0, 0, 0)
	end if
end event


// Dibuja una linea de MENU_2 (entrada object.value1) centrada en pantalla.
function TRHText_DrawLine
	GetTextInfo(temp3, MENU_2, TEXTINFO_TEXTSIZE, object.value1, 0)
	temp4 = temp3
	temp4 <<= 2
	FlipSign(temp4)
	temp4 += screen.xcenter
	temp4 += 4
	temp0 = 0
	while temp3 > 0
		GetTextInfo(temp1, MENU_2, TEXTINFO_TEXTDATA, object.value1, temp0)
		temp5 = 0
		// mayusculas A-Z -> frames 10..35 (mapeo identico a TextMessage)
		if temp1 > 64
			if temp1 < 91
				temp5 = temp1
				temp5 -= 55
			end if
		end if
		// el espacio (frame 0) no se dibuja pero avanza 8px
		if temp5 > 0
			DrawSpriteScreenXY(temp5, temp4, object.ypos)
		end if
		temp4 += 8
		temp0++
		temp3--
	loop
end function


event ObjectDraw
	if object.value0 < 360
		object.ypos = 92
		object.value1 = 0
		CallFunction(TRHText_DrawLine)
		object.ypos += 20
		object.value1 = 1
		CallFunction(TRHText_DrawLine)
		object.ypos += 20
		object.value1 = 2
		CallFunction(TRHText_DrawLine)
	end if
end event
