// Function declarations
reserve function BigMushroom_DebugDraw

// Static Values

// Tables

function BigMushroom_DebugDraw
	temp0 = object.direction
	temp0 &= 1
	DrawSprite(temp0)
end function


function BigMushroom_DebugSpawn
	CreateTempObject(TypeName[Big Mushroom], 0, object.xpos, object.ypos)
	object[tempObjectPos].drawOrder = 3
	object[tempObjectPos].propertyValue = object.direction
	object[tempObjectPos].propertyValue &= 1
end function


event ObjectDraw
	DrawSprite(0)
end event

	CheckCurrentStageFolder("Zone13")
	if checkResult == 1
		LoadSpriteSheet("WZ/Objects.gif")

		//   asi que aqui se carga esa hoja.

		SpriteFrame(-8, -8, 16, 16, 111, 333)
